Mã nguồn Thuật toán Tính điểm Minh bạch

function calculateTrustScore(safety, transparency, legality, redFlags) {
  const WEIGHT_SAFETY = 0.40;
  const WEIGHT_TRANSPARENCY = 0.30;
  const WEIGHT_LEGALITY = 0.30;

  // Tính điểm chuẩn theo trọng số minh bạch
  let baseScore = (safety * 0.40) + (transparency * 0.30) + (legality * 0.30);

  // Cơ chế Khóa Phạt Vi Phạm (Penalty Gate):
  // Bẫy lừa đảo hoặc cờ bạc bất hợp pháp bị khóa chặt ở mức điểm nguy hiểm
  if (redFlags.includes('PROHIBITED_GAMBLING') || redFlags.includes('ACTIVE_PHISHING')) {
    baseScore = Math.min(baseScore, 15);
  }
  return Math.round(baseScore);
}
